    <script>
        const canvas = document.getElementById('bg-canvas');
        const ctx = canvas.getContext('2d');
        const world = document.getElementById('physics-world');
        const overlay = document.getElementById('modal-overlay');
        const mTitle = document.getElementById('m-title');
        const mBody = document.getElementById('m-body');

        let width, height;
        const entities = [];
        const stars = [];

        const config = [
            { id: '3DCHEM', type: 'atom', label: '3D_CHEM', color: '#ff0044' },
            { id: 'TICTACTOE', type: 'console', label: 'TTT_LOGIC', color: '#4a4d5e' },
            { id: 'SOON_1', type: 'apple', label: 'NODE_01', color: '#d81159' },
            { id: 'SOON_2', type: 'orange', label: 'NODE_02', color: '#fb8500' },
            { id: 'SOON_3', type: 'pear', label: 'NODE_03', color: '#2ecc71' }
        ];

        function initStars() {
            stars.length = 0;
            const starCount = 150;
            for (let i = 0; i < starCount; i++) {
                const speed = Math.random() * 0.5 + 0.1;
                stars.push({
                    x: Math.random() * width,
                    y: Math.random() * height,
                    size: Math.random() * 1 + 0.5, // Smaller stars
                    speed: speed,
                    trailLen: speed * 5 // Minimal trail
                });
            }
        }

        function initEntities() {
            world.innerHTML = '';
            entities.length = 0;
            config.forEach((cfg, index) => {
                const el = document.createElement('div');
                el.className = 'entity';
                el.onclick = () => openModule(cfg.id);

                let content = '';
                if (cfg.type === 'atom') {
                    content = `<div class="pixel-atom"><div class="nucleus"></div><div class="orbit orbit-1"></div><div class="orbit orbit-2"></div></div>`;
                } else if (cfg.type === 'console') {
                    content = `<div class="pixel-console"><div class="console-screen"><div class="grid-cell active-x">X</div><div class="grid-cell"></div><div class="grid-cell"></div><div class="grid-cell"></div><div class="grid-cell active-x">X</div><div class="grid-cell"></div><div class="grid-cell"></div><div class="grid-cell"></div><div class="grid-cell active-x">X</div></div></div>`;
                } else {
                    content = `<div class="pixel-fruit ${cfg.type}"><div class="fruit-stem"></div></div>`;
                }

                el.innerHTML = `${content}<div class="label">${cfg.label}</div>`;
                world.appendChild(el);

                const size = cfg.type === 'atom' ? 80 : (cfg.type === 'console' ? 75 : 55);
                entities.push({
                    el: el,
                    id: cfg.id,
                    x: Math.random() * (width - 150) + 75,
                    y: Math.random() * (height - 150) + 75,
                    vx: (Math.random() - 0.5) * 3,
                    vy: (Math.random() - 0.5) * 3,
                    r: size / 2,
                    mass: size / 8
                });
            });
        }

        function render() {
            ctx.fillStyle = '#050508';
            ctx.fillRect(0, 0, width, height);

            // Draw Stars and Subtle Trails
            stars.forEach(star => {
                // Trail
                ctx.beginPath();
                ctx.strokeStyle = `rgba(255, 255, 255, 0.15)`;
                ctx.lineWidth = star.size;
                ctx.moveTo(star.x, star.y);
                ctx.lineTo(star.x + star.trailLen, star.y);
                ctx.stroke();

                // Core
                ctx.fillStyle = `rgba(255, 255, 255, 0.8)`;
                ctx.fillRect(star.x, star.y, star.size, star.size);

                star.x -= star.speed;
                if (star.x < -10) {
                    star.x = width + 10;
                    star.y = Math.random() * height;
                }
            });

            // Physics Update
            for (let i = 0; i < entities.length; i++) {
                const e = entities[i];
                e.x += e.vx;
                e.y += e.vy;

                if (e.x < e.r) { e.x = e.r; e.vx *= -1; }
                if (e.x > width - e.r) { e.x = width - e.r; e.vx *= -1; }
                if (e.y < e.r) { e.y = e.r; e.vy *= -1; }
                if (e.y > height - e.r) { e.y = height - e.r; e.vy *= -1; }

                for (let j = i + 1; j < entities.length; j++) {
                    const e2 = entities[j];
                    const dx = e2.x - e.x;
                    const dy = e2.y - e.y;
                    const dist = Math.sqrt(dx*dx + dy*dy);
                    const minDist = e.r + e2.r;

                    if (dist < minDist) {
                        const nx = dx / dist;
                        const ny = dy / dist;
                        const overlap = minDist - dist;
                        e.x -= nx * (overlap / 2);
                        e.y -= ny * (overlap / 2);
                        e2.x += nx * (overlap / 2);
                        e2.y += ny * (overlap / 2);

                        const p = 2 * (e.vx * nx + e.vy * ny - e2.vx * nx - e2.vy * ny) / (e.mass + e2.mass);
                        e.vx -= p * e2.mass * nx;
                        e.vy -= p * e2.mass * ny;
                        e2.vx += p * e.mass * nx;
                        e2.vy += p * e.mass * ny;
                    }
                }
                e.el.style.transform = `translate(${e.x - e.r}px, ${e.y - e.r}px)`;
            }

            requestAnimationFrame(render);
        }

        function openModule(id) {
            const data = {
                '3DCHEM': { t: '3D_CHEM_LATTICE', b: 'MOLECULAR SCAN COMPLETE. ATOMIC BONDS STABLE. 3D VISUALIZER ACTIVE.' },
                'TICTACTOE': { t: 'LOGIC_CONSOLE_v1', b: 'STRATEGIC GRID DETECTED. AWAITING PLAYER ONE INPUT.' },
                'DEFAULT': { t: 'DATA_UNKNOWN', b: 'COSMIC FRAGMENT UNREADABLE. RIPENING PROCESS INCOMPLETE.' }
            };
            const content = data[id] || data['DEFAULT'];
            mTitle.innerText = content.t;
            mBody.innerText = content.b;
            overlay.style.display = 'flex';
        }

        function closeModule() { overlay.style.display = 'none'; }

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
            initStars();
        }

        window.onload = () => {
            resize();
            initEntities();
            render();
        };
        window.onresize = resize;
    </script>
